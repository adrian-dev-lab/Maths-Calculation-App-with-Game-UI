class _PremiumSparkle extends StatelessWidget {
  final double size;
  final Color color;

  const _PremiumSparkle({
    required this.size,
    required this.color,
  });

  @override
  Widget build(BuildContext context) {
    return CustomPaint(
      size: Size.square(size),
      painter: _PremiumSparklePainter(color),
    );
  }
}


class _PremiumSparklePainter extends CustomPainter {
  final Color color;

  _PremiumSparklePainter(this.color);

  @override
  void paint(Canvas canvas, Size size) {
    final cx = size.width / 2;
    final cy = size.height / 2;

    final path = Path()
      ..moveTo(cx, 0)
      ..quadraticBezierTo(
        cx + size.width * 0.09,
        cy - size.height * 0.09,
        size.width,
        cy,
      )
      ..quadraticBezierTo(
        cx + size.width * 0.09,
        cy + size.height * 0.09,
        cx,
        size.height,
      )
      ..quadraticBezierTo(
        cx - size.width * 0.09,
        cy + size.height * 0.09,
        0,
        cy,
      )
      ..quadraticBezierTo(
        cx - size.width * 0.09,
        cy - size.height * 0.09,
        cx,
        0,
      )
      ..close();

    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.fill;

    canvas.drawShadow(
      path,
      color.withValues(alpha: 0.42),
      6,
      false,
    );

    canvas.drawPath(path, paint);
  }

  @override
  bool shouldRepaint(
    covariant _PremiumSparklePainter oldDelegate,
  ) {
    return oldDelegate.color != color;
  }
}
