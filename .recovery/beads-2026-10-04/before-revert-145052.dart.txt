enum FlashState { idle, flashing, question, answer }

  const BeadsScreen({super.key, required this.onBack});

  @override
  State<BeadsScreen> createState() => _BeadsScreenState();

class _BeadsScreenState extends State<BeadsScreen> {
  late int sequenceLength;
  late String speed;
  late int digitsCount;
  late String operation;

  FlashState currentState = FlashState.idle;
  List<int> numbers = [];
  int currentFlashIndex = 0;
  bool isFlashingVisible = false;
  int totalSum = 0;
  Timer? flashTimer;
  double baseFontSize = 24.0;

  @override
  void initState() {
    super.initState();
    sequenceLength = StorageService.loadBeadsSequence();
    speed = StorageService.loadBeadsSpeed();
    digitsCount = StorageService.loadBeadsGroup();
    operation = StorageService.loadBeadsOperation();
    _loadFontSize();
  }

  void _loadFontSize() {
    final settings = StorageService.loadSettings();
    switch (settings.fontSize) {
      case DisplayFontSize.extraSmall:
        baseFontSize = 12.0;
        break;
      case DisplayFontSize.small:
        baseFontSize = 16.0;
        break;
      case DisplayFontSize.normal:
        baseFontSize = 24.0;
        break;
      case DisplayFontSize.large:
        baseFontSize = 32.0;
        break;
      case DisplayFontSize.extraLarge:
        baseFontSize = 48.0;
        break;
    }
  }

  void _showSettings() {
    if (currentState != FlashState.idle) {
      return; // Prevent settings while playing
    }
    Navigator.push(
      context,
      MaterialPageRoute(
        builder: (context) => BeadsSettingsScreen(
          initialColumnCount: sequenceLength,
          initialSpeed: speed,
          initialGroup: digitsCount,
          initialOperation: operation,
          onSave: (newSequence, newSpeed, newGroup, newOperation) {
            StorageService.saveBeadsSequence(newSequence);
            StorageService.saveBeadsSpeed(newSpeed);
            StorageService.saveBeadsGroup(newGroup);
            StorageService.saveBeadsOperation(newOperation);
            setState(() {
              sequenceLength = newSequence;
              speed = newSpeed;
              digitsCount = newGroup;
              operation = newOperation;
            });
          },
        ),
      ),
    );
  }

  void _startFlash() {
    // Generate random numbers
    final rand = math.Random();
    int minVal = digitsCount == 1 ? 1 : math.pow(10, digitsCount - 1).toInt();
    int maxVal = math.pow(10, digitsCount).toInt() - 1;

    numbers = [];
    int currentSum = 0;

    for (int i = 0; i < sequenceLength; i++) {
      int val = minVal + rand.nextInt(maxVal - minVal + 1);

      bool makeNegative = false;
      if (i > 0) {
        if (operation == 'Subtraction') {
          makeNegative = true;
        } else if (operation == 'Mixed') {
          makeNegative = rand.nextBool();
        }
      }

      if (makeNegative) {
        if (currentSum >= minVal) {
          int maxSub = math.min(currentSum, maxVal);
          val = -(minVal + rand.nextInt(maxSub - minVal + 1));
        } else {
          // If we can't subtract without going below 0 (and below minVal digits), just add
          val = val;
        }
      }

      numbers.add(val);
      currentSum += val;
    }
    totalSum = currentSum;

    setState(() {
      currentState = FlashState.flashing;
      currentFlashIndex = 0;
      isFlashingVisible = true;
    });

    int delayMs = speed == 'Fast' ? 600 : (speed == 'Slow' ? 1800 : 1200);
    int showMs = (delayMs * 0.7).toInt();
    int hideMs = delayMs - showMs;

    _runFlashLoop(showMs, hideMs);
  }

  void _runFlashLoop(int showMs, int hideMs) {
    if (currentFlashIndex >= numbers.length) {
      Future.delayed(const Duration(milliseconds: 300), () {
        if (mounted) {
          setState(() {
            currentState = FlashState.question;
          });
        }
      });
      return;
    }

    if (mounted) {
      setState(() {
        isFlashingVisible = true;
      });
    }

    flashTimer = Timer(Duration(milliseconds: showMs), () {
      if (mounted) {
        setState(() {
          isFlashingVisible = false;
        });

        flashTimer = Timer(Duration(milliseconds: hideMs), () {
          currentFlashIndex++;
          _runFlashLoop(showMs, hideMs);
        });
      }
    });
  }

  @override
  void dispose() {
    flashTimer?.cancel();
    super.dispose();
  }

class _BeadsAnswerResult extends StatelessWidget {
  final int answer;
  final VoidCallback onPlayAgain;

  const _BeadsAnswerResult({
    required this.answer,
    required this.onPlayAgain,
  });

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(
      builder: (context, constraints) {
        final screenSize = MediaQuery.sizeOf(context);

        final availableWidth = constraints.maxWidth.isFinite
            ? constraints.maxWidth
            : screenSize.width;

        // Keep the visual composition compact on Windows/tablet.
        final contentWidth =
            (availableWidth * 0.72).clamp(340.0, 620.0);

        final cardHeight =
            (contentWidth * 0.40).clamp(170.0, 220.0);

        final titleSize =
            (contentWidth * 0.095).clamp(36.0, 54.0);

        final answerFontSize =
            (contentWidth * 0.25).clamp(96.0, 150.0);

        final buttonHeight =
            (contentWidth * 0.13).clamp(60.0, 78.0);

        final formattedAnswer = answer.toString().replaceAllMapped(
          RegExp(r'(\d{1,3})(?=(\d{3})+(?!\d))'),
          (match) => '${match[1]},',
        );

        // Empty hero area for the exact trophy asset we will add later.

        return Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.symmetric(vertical: 20),
            child: SizedBox(
              width: contentWidth,
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  const _ResultHero(),

                  Text(
                    'Answer',
                    textAlign: TextAlign.center,
                    style: TextStyle(
                      fontSize: titleSize,
                      height: 1,
                      fontWeight: FontWeight.w800,
                      color: const Color(0xFFF6F1FF),
                      shadows: const [
                        Shadow(
                          color: Color(0x33000000),
                          blurRadius: 6,
                          offset: Offset(0, 3),
                        ),
                      ],
                    ),
                  ),

                  SizedBox(
                    height: (contentWidth * 0.055).clamp(18.0, 30.0),
                  ),

                  _AnswerCard(
                    formattedAnswer: formattedAnswer,
                    height: cardHeight,
                    answerFontSize: answerFontSize,
                  ),

                  SizedBox(
                    height: (contentWidth * 0.08).clamp(28.0, 44.0),
                  ),

                  SizedBox(
                    width: contentWidth * 0.96,
                    child: _PrimaryGradientButton(
                      height: buttonHeight,
                      onPressed: onPlayAgain,
                    ),
                  ),
                ],
              ),
            ),
          ),
        );
      },
    );
  }
}

class _TrophyCupClipper extends CustomClipper<Path> {
  @override
  Path getClip(Size size) {
    final path = Path();

    path.moveTo(5, 0);

    // top left
    path.quadraticBezierTo(
      2,
      size.height * 0.10,
      7,
      size.height * 0.25,
    );

    // left body taper
    path.cubicTo(
      size.width * 0.08,
      size.height * 0.58,
      size.width * 0.24,
      size.height * 0.88,
      size.width * 0.50,
      size.height,
    );

    // right mirrored curve
    path.cubicTo(
      size.width * 0.76,
      size.height * 0.88,
      size.width * 0.92,
      size.height * 0.58,
      size.width - 7,
      size.height * 0.25,
    );

    path.quadraticBezierTo(
      size.width - 2,
      size.height * 0.10,
      size.width - 5,
      0,
    );

    path.close();

    return path;
  }

  @override
  bool shouldReclip(
    covariant CustomClipper<Path> oldClipper,
  ) =>
      false;
}


// ============================================================
// Trophy star
// ============================================================

class _GlossyTrophyStarPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    const points = 5;

    final center =
        Offset(size.width / 2, size.height / 2);

    final outerRadius = size.width * 0.44;
    final innerRadius = outerRadius * 0.45;

    final path = Path();

    for (int i = 0; i < points * 2; i++) {
      final radius =
          i.isEven ? outerRadius : innerRadius;

      final angle =
          -math.pi / 2 +
          i * math.pi / points;

      final p = Offset(
        center.dx +
            math.cos(angle) * radius,
        center.dy +
            math.sin(angle) * radius,
      );

      if (i == 0) {
        path.moveTo(p.dx, p.dy);
      } else {
        path.lineTo(p.dx, p.dy);
      }
    }

    path.close();

    final rect = Rect.fromLTWH(
      0,
      0,
      size.width,
      size.height,
    );

    final starPaint = Paint()
      ..shader = const LinearGradient(
        begin: Alignment.topLeft,
        end: Alignment.bottomRight,
        colors: [
          Color(0xFFFFF178),
          Color(0xFFFFC320),
          Color(0xFFFF8E00),
        ],
      ).createShader(rect);

    canvas.drawShadow(
      path,
      const Color(0xFFB15A00),
      6,
      false,
    );

    canvas.drawPath(
      path,
      starPaint,
    );

    // Small glossy upper highlight
    canvas.save();
    canvas.clipPath(path);

    final highlightPaint = Paint()
      ..shader = LinearGradient(
        begin: Alignment.topCenter,
        end: Alignment.bottomCenter,
        colors: [
          Colors.white.withValues(alpha: 0.48),
          Colors.white.withValues(alpha: 0.00),
        ],
      ).createShader(
        Rect.fromLTWH(
          0,
          0,
          size.width,
          size.height * 0.55,
        ),
      );

    canvas.drawRect(
      Rect.fromLTWH(
        0,
        0,
        size.width,
        size.height * 0.55,
      ),
      highlightPaint,
    );

    canvas.restore();
  }

  @override
  bool shouldRepaint(
    covariant CustomPainter oldDelegate,
  ) =>
      false;
}

class _FourPointSparkle extends StatelessWidget {
  final double size;
  final Color color;

  const _FourPointSparkle({
    required this.size,
    required this.color,
  });

  @override
  Widget build(BuildContext context) {
    return CustomPaint(
      size: Size.square(size),
      painter: _FourPointSparklePainter(color),
    );
  }
}


class _FourPointSparklePainter
    extends CustomPainter {
  final Color color;

  _FourPointSparklePainter(this.color);

  @override
  void paint(Canvas canvas, Size size) {
    final cx = size.width / 2;
    final cy = size.height / 2;

    final path = Path()
      ..moveTo(cx, 0)
      ..quadraticBezierTo(
        cx + size.width * 0.09,
        cy - size.height * 0.09,
        size.width,
        cy,
      )
      ..quadraticBezierTo(
        cx + size.width * 0.09,
        cy + size.height * 0.09,
        cx,
        size.height,
      )
      ..quadraticBezierTo(
        cx - size.width * 0.09,
        cy + size.height * 0.09,
        0,
        cy,
      )
      ..quadraticBezierTo(
        cx - size.width * 0.09,
        cy - size.height * 0.09,
        cx,
        0,
      )
      ..close();

    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.fill;

    canvas.drawShadow(
      path,
      color.withValues(alpha: 0.35),
      6,
      false,
    );

    canvas.drawPath(path, paint);
  }

  @override
  bool shouldRepaint(
    covariant CustomPainter oldDelegate,
  ) =>
      false;
}

        value.toString().padLeft(columnCount, '0');

        .substring(
          text.length > columnCount

              : 0,

        .map((e) => int.tryParse(e) ?? 0)
        .toList();

// ============================================================
// Single Soroban column
//
// IMPORTANT FOR COLOR-BLIND ACCESSIBILITY:
// Value is communicated by POSITION.
// Colors are decorative only.
//
// Digit rule:
// 0–4 => upper bead away from beam
// 5–9 => upper bead touches beam
//
// lower bead count touching beam = digit % 5
// ============================================================

class _SorobanColumn extends StatelessWidget {
  final int digit;
  final double beadWidth;
  final double beadHeight;
  final double rodWidth;
  final double totalHeight;
  final double beamHeight;

  const _SorobanColumn({
    required this.digit,
    required this.beadWidth,
    required this.beadHeight,
    required this.rodWidth,
    required this.totalHeight,
    required this.beamHeight,
  });

  @override
  Widget build(BuildContext context) {
    final upperActive = digit >= 5;
    final lowerActive = digit % 5;

    final beadGap = beadHeight * 0.12;
    final beadStep = beadHeight + beadGap;

    final beamY = totalHeight * 0.315;

    final upperInactiveTop =
        beamY - (beadHeight * 2.0);

    final upperActiveTop =
        beamY - beadHeight - 6;

    final lowerActiveStart =
        beamY + beamHeight + 6;

    final lowerInactiveBottomMargin =
        totalHeight * 0.045;

    final inactiveCount = 4 - lowerActive;

    final inactiveStackHeight =
        inactiveCount > 0
            ? inactiveCount * beadHeight +
              (inactiveCount - 1) * beadGap
            : 0.0;

    final inactiveStart =
        totalHeight -
        lowerInactiveBottomMargin -
        inactiveStackHeight;

    return Stack(
      alignment: Alignment.topCenter,
      children: [
        // ------------------------------------------------------
        // Vertical rod
        // ------------------------------------------------------
        Positioned(
          top: totalHeight * 0.035,
          bottom: totalHeight * 0.035,
          child: Container(
            width: rodWidth,
            decoration: BoxDecoration(
              borderRadius:
                  BorderRadius.circular(100),
              gradient: const LinearGradient(
                begin: Alignment.centerLeft,
                end: Alignment.centerRight,
                colors: [
                  Color(0xFFFFD58A),
                  Color(0xFFE58B3B),
                  Color(0xFF9D4A1B),
                ],
              ),
              boxShadow: [
                BoxShadow(
                  color: Colors.black
                      .withValues(alpha: 0.24),
                  blurRadius: 5,
                ),
              ],
            ),
          ),
        ),

        // ------------------------------------------------------
        // Upper bead (worth 5)
        // Gold is decorative.
        // Position tells whether the bead is active.
        // ------------------------------------------------------
        AnimatedPositioned(
          duration:
              const Duration(milliseconds: 150),
          curve: Curves.easeOutCubic,
          top: upperActive
              ? upperActiveTop
              : upperInactiveTop,
          child: _GoldUpperBead(
            width: beadWidth,
            height: beadHeight,
          ),
        ),

        // ------------------------------------------------------
        // Lower beads
        // 4 beads worth 1 each.
        //
        // Active beads move toward beam.
        // Inactive beads remain lower.
        // ------------------------------------------------------
        for (int i = 0; i < 4; i++)
          AnimatedPositioned(
            duration:
                const Duration(milliseconds: 150),
            curve: Curves.easeOutCubic,
            top: i < lowerActive
                ? lowerActiveStart + (i * beadStep)
                : inactiveStart + ((i - lowerActive) * beadStep),
            child: _PurpleLowerBead(
              width: beadWidth,
              height: beadHeight,
            ),
          ),
      ],
    );
  }
}

// ============================================================
// Faceted bead painter
// ============================================================

class _FacetedBeadPainter
    extends CustomPainter {
  final Color topColor;
  final Color middleColor;
  final Color bottomColor;
  final Color outlineColor;
  final bool strongOutline;

  _FacetedBeadPainter({
    required this.topColor,
    required this.middleColor,
    required this.bottomColor,
    required this.outlineColor,
    required this.strongOutline,
  });

  @override
  void paint(
    Canvas canvas,
    Size size,
  ) {
    final w = size.width;
    final h = size.height;

    final path = Path()
      ..moveTo(w * 0.22, 0)
      ..lineTo(w * 0.78, 0)
      ..quadraticBezierTo(
        w * 0.88,
        h * 0.03,
        w,
        h * 0.44,
      )
      ..quadraticBezierTo(
        w * 0.96,
        h * 0.70,
        w * 0.77,
        h,
      )
      ..lineTo(w * 0.23, h)
      ..quadraticBezierTo(
        w * 0.04,
        h * 0.70,
        0,
        h * 0.44,
      )
      ..quadraticBezierTo(
        w * 0.12,
        h * 0.03,
        w * 0.22,
        0,
      )
      ..close();

    final rect =
        Rect.fromLTWH(0, 0, w, h);

    final fill = Paint()
      ..shader = LinearGradient(
        begin: Alignment.topCenter,
        end: Alignment.bottomCenter,
        colors: [
          topColor,
          middleColor,
          bottomColor,
        ],
        stops: const [
          0.0,
          0.48,
          1.0,
        ],
      ).createShader(rect);

    canvas.drawShadow(
      path,
      Colors.black.withValues(alpha: 0.40),
      8,
      false,
    );

    canvas.drawPath(path, fill);

    // --------------------------------------------------------
    // Top-left faceted highlight
    // --------------------------------------------------------
    final highlight = Path()
      ..moveTo(w * 0.22, h * 0.06)
      ..lineTo(w * 0.51, h * 0.06)
      ..lineTo(w * 0.39, h * 0.45)
      ..lineTo(w * 0.10, h * 0.45)
      ..close();

    final highlightPaint = Paint()
      ..color =
          Colors.white.withValues(alpha: 0.22);

    canvas.drawPath(
      highlight,
      highlightPaint,
    );

    // --------------------------------------------------------
    // Bottom facet
    // --------------------------------------------------------
    final bottomFacet = Path()
      ..moveTo(0, h * 0.50)
      ..lineTo(w, h * 0.50)
      ..lineTo(w * 0.77, h)
      ..lineTo(w * 0.23, h)
      ..close();

    final bottomPaint = Paint()
      ..color = Colors.black.withValues(
        alpha: 0.08,
      );

    canvas.drawPath(
      bottomFacet,
      bottomPaint,
    );

    // --------------------------------------------------------
    // Accessibility outline
    // --------------------------------------------------------
    final outline = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth =
          strongOutline ? 2.0 : 1.2
      ..color = outlineColor.withValues(
        alpha: strongOutline ? 0.75 : 0.35,
      );

    canvas.drawPath(
      path,
      outline,
    );
  }

  @override
  bool shouldRepaint(
    covariant _FacetedBeadPainter
        oldDelegate,
  ) {
    return oldDelegate.topColor != topColor ||
        oldDelegate.middleColor !=
            middleColor ||
        oldDelegate.bottomColor !=
            bottomColor ||
        oldDelegate.outlineColor !=
            outlineColor ||
        oldDelegate.strongOutline !=
            strongOutline;
  }
}

// ============================================================
// SPARKLE
// ============================================================

class _FinalHeroSparkle extends StatelessWidget {
  final double size;
  final Color color;

  const _FinalHeroSparkle({
    required this.size,
    required this.color,
  });

  @override
  Widget build(BuildContext context) {
    return CustomPaint(
      size: Size.square(size),
      painter: _FinalHeroSparklePainter(color),
    );
  }
}


class _FinalHeroSparklePainter extends CustomPainter {
  final Color color;

  _FinalHeroSparklePainter(this.color);

  @override
  void paint(Canvas canvas, Size size) {
    final cx = size.width / 2;
    final cy = size.height / 2;

    final path = Path()
      ..moveTo(cx, 0)
      ..quadraticBezierTo(
        cx + size.width * 0.09,
        cy - size.height * 0.09,
        size.width,
        cy,
      )
      ..quadraticBezierTo(
        cx + size.width * 0.09,
        cy + size.height * 0.09,
        cx,
        size.height,
      )
      ..quadraticBezierTo(
        cx - size.width * 0.09,
        cy + size.height * 0.09,
        0,
        cy,
      )
      ..quadraticBezierTo(
        cx - size.width * 0.09,
        cy - size.height * 0.09,
        cx,
        0,
      )
      ..close();

    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.fill;

    canvas.drawShadow(
      path,
      color.withValues(alpha: 0.40),
      6,
      false,
    );

    canvas.drawPath(path, paint);
  }

  @override
  bool shouldRepaint(
    covariant _FinalHeroSparklePainter oldDelegate,
  ) {
    return oldDelegate.color != color;
  }
}
